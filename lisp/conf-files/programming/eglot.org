* Dape-Repeat
#+begin_src emacs-lisp
(when (fboundp 'repeat-mode)
  ;; Включаем repeat-mode глобально
  (repeat-mode +1)

  ;; Карта повторения для отладочных команд
  (defvar dape-repeat-map
    (let ((map (make-sparse-keymap)))
      ;; Шаги и выполнение
      (define-key map "n" #'dape-next)
      (define-key map "i" #'dape-step-in)
      (define-key map "o" #'dape-step-out)
      (define-key map "c" #'dape-continue)
      (define-key map "p" #'dape-pause)
      (define-key map "u" #'dape-until)

      ;; Брейкпоинты
      (define-key map "b" #'dape-breakpoint-toggle)

      ;; Сессия
      (define-key map "r" #'dape-restart)
      (define-key map "k" #'dape-kill)
      (define-key map "q" #'dape-quit)
      (define-key map "D" #'dape-disconnect-quit)
      (define-key map "s" #'dape-select-session)

      ;; Стек / потоки
      (define-key map "t" #'dape-select-thread)
      (define-key map "S" #'dape-select-stack)

      ;; Инспекция
      (define-key map "e" #'dape-evaluate-expression)
      (define-key map "w" #'dape-watch-dwim)
      (define-key map "m" #'dape-memory)
      (define-key map "d" #'dape-disassemble)
      map)
    "Repeat map for `dape' debugging commands.")

  ;; Привязываем свойство repeat-map к командам.
  ;; Именно это заставляет repeat-mode активировать карту
  ;; после выполнения любой из этих команд.
  (dolist (cmd '(dape-next dape-step-in dape-step-out dape-continue
                 dape-pause dape-until dape-breakpoint-toggle
                 dape-restart dape-kill dape-quit dape-disconnect-quit
                 dape-select-session dape-select-thread dape-select-stack
                 dape-evaluate-expression dape-watch-dwim
                 dape-memory dape-disassemble))
    (put cmd 'repeat-map 'dape-repeat-map)))
#+end_src
