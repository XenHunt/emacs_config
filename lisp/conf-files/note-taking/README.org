При помощи *qwen3.8* получилось сделать более продуманный вариант задач для каждого из проектов.

#+begin_src emacs-lisp
(defcustom my/org-roam-project-tasks-heading "Tasks"
  "Название корневого заголовка задач в проекте."
  :type 'string)

(defvar my/org-roam-task-parent-history nil
  "История выбора родительской задачи для новой задачи.")

(defun my/org-roam--ensure-project-tasks-heading (file)
  "Убедиться, что в FILE есть корневой заголовок задач.
Название заголовка берётся из `my/org-roam-project-tasks-heading'."
  (when (and file (file-exists-p file))
    (with-current-buffer (find-file-noselect file)
      (widen)
      (unless (derived-mode-p 'org-mode)
        (org-mode))
      (goto-char (point-min))
      (unless (org-find-exact-headline-in-buffer
               my/org-roam-project-tasks-heading nil t)
        (goto-char (point-max))
        (unless (bolp) (insert "\n"))
        (insert "\n* " my/org-roam-project-tasks-heading "\n")))))

(defun my/org-roam--unique-label (label seen)
  "Сделать LABEL уникальным, используя хэш-таблицу SEEN."
  (let ((base label)
        (n 2))
    (while (gethash label seen)
      (setq label (format "%s [%d]" base n)
            n (1+ n)))
    (puthash label t seen)
    label))

(defun my/org-roam--format-task-label (plist)
  "Человекочитаемый подписи для кандидата задачи.
PLIST содержит :olp, :level и :root-level."
  (let* ((olp (plist-get plist :olp))
         (level (plist-get plist :level))
         (root-level (plist-get plist :root-level)))
    (if (= level root-level)
        (format "%s (корень задач)" (car olp))
      (concat
       (make-string (* 2 (- level root-level)) ?\s)
       (mapconcat #'identity olp " › ")))))

(defun my/org-roam--project-task-candidates (file)
  "Список кандидатов вида (LABEL . PLIST) для заголовков внутри `Tasks' в FILE.

Каждый PLIST содержит:
  :olp        — outline path до заголовка;
  :level      — уровень заголовка;
  :root-level — уровень корневого заголовка задач."
  (let ((candidates '())
        (seen (make-hash-table :test #'equal)))
    (with-current-buffer (find-file-noselect file)
      (widen)
      (unless (derived-mode-p 'org-mode)
        (org-mode))
      (goto-char (point-min))
      (if-let* ((pos (org-find-exact-headline-in-buffer
                      my/org-roam-project-tasks-heading nil t)))
          (progn
            (goto-char pos)
            (let* ((root-level (org-current-level))
                   (root-marker (point-marker))
                   (root-end (save-excursion
                               (org-end-of-subtree t)
                               (point-marker))))
              ;; Сам корень задач.
              (push (list :olp (mapcar #'substring-no-properties
                                       (org-get-outline-path t))
                          :level root-level
                          :root-level root-level)
                    candidates)

              ;; Все заголовки внутри поддерева Tasks.
              (org-map-entries
               (lambda ()
                 (when (and (> (point) root-marker)
                            (< (point) root-end)
                            (> (org-current-level) root-level))
                   (push (list :olp (mapcar #'substring-no-properties
                                            (org-get-outline-path t))
                               :level (org-current-level)
                               :root-level root-level)
                         candidates)))
               nil nil)))
        ;; Если заголовка задач ещё нет, даём виртуальный корень.
        (push (list :olp (list my/org-roam-project-tasks-heading)
                    :level 1
                    :root-level 1)
              candidates)))

    (setq candidates (nreverse candidates))
    (mapcar
     (lambda (plist)
       (let ((label (my/org-roam--format-task-label plist)))
         (cons (my/org-roam--unique-label label seen) plist)))
     candidates)))

(defun my/org-roam--select-project-task-parent (file)
  "Выбрать родительский заголовок внутри `Tasks' в FILE."
  (let* ((candidates (my/org-roam--project-task-candidates file)))
    (if (= 1 (length candidates))
        ;; Если внутри только корень задач — не спрашиваем зря.
        (cdar candidates)
      (let* ((labels (mapcar #'car candidates))
             (choice (if (fboundp 'consult--read)
                         (consult--read
                          labels
                          :prompt "К какой задаче прикрепить новую: "
                          :require-match t
                          :sort nil
                          :history 'my/org-roam-task-parent-history)
                       (completing-read
                        "К какой задаче прикрепить новую: "
                        labels nil t nil
                        'my/org-roam-task-parent-history))))
        (cdr (assoc choice candidates))))))

(defun my/org-roam-capture-task ()
  "Создать задачу в Org-Roam-проекте и прикрепить её к выбранному родителю.

Родителя можно выбрать среди заголовков внутри корневого `Tasks'.
Таким образом можно строить произвольное дерево задач"
  (interactive)
  (let ((node (org-roam-node-read nil (my/org-roam-filter-by-tag "Project"))))
    (unless node
      (user-error "Проект не выбран"))
    (let* ((file (org-roam-node-file node))
           (existing-file (and file (file-exists-p file))))
      (let* ((parent (if existing-file
                         (my/org-roam--select-project-task-parent file)
                       (list :olp (list my/org-roam-project-tasks-heading)
                             :level 1
                             :root-level 1)))
             (olp (plist-get parent :olp))
             (level (or (plist-get parent :level) 1))
             (task-heading (concat (make-string (1+ level) ?*) " TODO %?"))
             (new-project-head (concat "#+title: ${title}\n"
                                       "#+category: ${title}\n"
                                       "#+filetags: Project\n"
                                       "* Описание\n"
                                       "%^{Описание}\n"
                                       "* " my/org-roam-project-tasks-heading "\n"))
             (templates
              (if existing-file
                  `(("t" "project task" plain ,task-heading
                     :target (file+olp ,file ,olp)
                     :unnarrowed t))
                `(("t" "project task" plain ,task-heading
                   :target (file+head+olp "%<%Y-%m-%d-%H:%M:%S>-${slug}.org"
                                          ,new-project-head
                                          ,olp)
                   :unnarrowed t)))))
        ;; Если заголовка `Tasks' в существующем проекте вдруг нет — создадим его.
        (when existing-file
          (my/org-roam--ensure-project-tasks-heading file))

        (add-hook 'org-capture-after-finalize-hook
                  #'my/org-roam-project-finalize-hook)
        (org-roam-capture- :node node :templates templates)))))

(defun my/citar-get-file (citekey)
  "Get the file path for CITEKEY, with user selection if multiple."
  (let* ((ht (citar-get-files citekey))
         (files (gethash citekey ht)))
    (cond
     ((null files)
      (error "No files found for citekey %s" citekey)
      (org-capture-kill))
     ((> (length files) 1)
      (let ((unique-files (delete-dups files))) ; Remove duplicates
        (if (= (length unique-files) 1)
            (car unique-files)
          (completing-read "Выбери pdf файл, на который мы будем ссылаться:" unique-files))))
     ((= (length files) 1) (car files)))))
